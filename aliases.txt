# ---------------------
# Remote / thin client
# ---------------------
# Attach to a persistent remote session. mosh survives roaming and suspend;
# tmux survives mosh dying. You want both, always, in that order.
# Usage: rdev myhost
rdev() { mosh "$1" -- tmux new-session -A -s main; }

# THE GOTCHA: mosh cannot forward ports. There is no `mosh -L`. So a dev server
# on the remote box is NOT reachable just because you're mosh'd in.
# Two ways round it, and the first is better:
#   1. Tailscale - just hit the box directly: http://myhost:3000. No tunnel to
#      babysit, works from the phone too. This is why Tailscale is worth having.
#   2. A separate autossh tunnel alongside the mosh session, for when the port
#      is bound to the remote 127.0.0.1 and you can't rebind it.
# Usage: tun myhost 3000
tun() { autossh -M 0 -N -L "$2:localhost:$2" "$1"; }

# Mount a remote project locally and edit it in local nvim (no latency on
# keystrokes; the build still runs remotely over mosh).
# Usage: rmount myhost /home/me/project  ->  ~/remote/project
rmount() { mkdir -p "$HOME/remote/$(basename "$2")"; sshfs "$1:$2" "$HOME/remote/$(basename "$2")"; }
rumount() { fusermount3 -u "$HOME/remote/$1" 2>/dev/null || fusermount -u "$HOME/remote/$1"; }

# Run a single remote GUI app in a local window (Wayland-native forwarding).
# Fine for editors and tools; falls apart on anything pushing video frames -
# use Moonlight for Godot playtests and video scrubbing.
# Usage: rgui myhost kdenlive
rgui() { waypipe ssh "$1" "$2"; }
